# Runtime dependencies for AN Telemetry.
#
# Versions are pinned to the combination that is tested on Python 3.11-3.14.
# Install into the project virtual environment (root .venv), not a random
# system Python:
#
#   python3 -m venv .venv
#   .venv/bin/python -m pip install -r backend/requirements.txt
#
# cryptography is a direct runtime dependency: app/zte_check.py signs the
# ZTE router's "Check" header with RSA PKCS#1 v1.5. Without it, router login
# and Access Control calls fail even though the rest of the app starts.
fastapi==0.162.0
uvicorn[standard]==0.54.0
sqlmodel==0.0.48
pydantic==2.14.2
scapy==2.8.0
httpx==0.28.1
python-dotenv==1.2.4
itsdangerous==2.2.0
psutil==7.2.2
cryptography==50.0.2
# Publishes netdash.local over mDNS so phones can open the dashboard without
# knowing the laptop's DHCP address. Needs UDP 5353 (unprivileged here).
zeroconf==0.154.0
# Provides the `speedtest` module imported by app/speedtest.py for the manual
# Speedtest page. Imported lazily, so a missing package only breaks that page -
# but it is still a runtime dependency and must be installed with the rest.
speedtest-cli==2.1.3

# uvicorn[standard] already installs websockets, but backend tests import it
# directly, so keep it explicit and pinned.
websockets==17.4.0
